make_bold_names <- function(mat, rc_fun, rc_names) {
  bold_names <- rc_fun(mat)
  ids <- rc_names %>% match(rc_fun(mat))
  ids %>%
    walk(
      function(i)
        bold_names[i] <<-
        bquote(italic(.(rc_fun(mat)[i]))) %>%
        as.expression()
    )
  bold_names
}

deg_list_gsea <- list()
for (ct in celltypes) {
  message("Running differential expression: ", ct)
  pbmc_ct <- subset(
    pbmc4,
    subset = celltype == ct
  )

  Idents(pbmc_ct) <- "groups"

  n_group <- table(Idents(pbmc_ct))
  print(n_group)

  if (!all(c("PBS", "LPS") %in% names(n_group))) {
    message("Skip ", ct, ": missing PBS or LPS")
    next
  }

  if (any(n_group[c("PBS", "LPS")] < 3)) {
    message("Skip ", ct, ": too few cells")
    next
  }

  deg_ct <- FindMarkers(
    object = pbmc_ct,
    ident.1 = "LPS",
    ident.2 = "PBS",
    only.pos = FALSE,
    min.pct = 0,
    logfc.threshold = 0
  )

  deg_ct$gene <- rownames(deg_ct)
  deg_ct$celltype <- ct

  deg_list_gsea[[ct]] <- deg_ct
}

print(sapply(deg_list_gsea, nrow))


# 3
gsea_list_all <- list()
gsea_list_simplified <- list()

for (ct in names(deg_list_gsea)) {

  message("==============================")
  message("Running GO-GSEA: ", ct)
  message("==============================")

  deg_ct <- deg_list_gsea[[ct]]

  if ("avg_log2FC" %in% colnames(deg_ct)) {
    fc_col <- "avg_log2FC"
  } else if ("avg_logFC" %in% colnames(deg_ct)) {
    fc_col <- "avg_logFC"
  } else {
    stop("Cannot find logFC column: ", ct)
  }

  rank_df <- deg_ct %>%
    transmute(
      gene = as.character(gene),
      score = .data[[fc_col]]
    ) %>%
    filter(
      !is.na(gene),
      gene != "",
      !is.na(score),
      is.finite(score)
    ) %>%
    group_by(gene) %>%
    slice_max(
      order_by = abs(score),
      n = 1,
      with_ties = FALSE
    ) %>%
    ungroup()

  rank_df <- rank_df %>%
    arrange(desc(score), gene)

  geneList <- rank_df$score
  names(geneList) <- rank_df$gene

  message("Genes used for GSEA: ", length(geneList))

  if (length(geneList) < 100) {
    message("Skip ", ct, ": too few genes")
    next
  }

  gsea_ct <- tryCatch(
    gseGO(
      geneList = geneList,
      OrgDb = org.Mm.eg.db,
      keyType = "SYMBOL",
      ont = "BP",
      minGSSize = 10,
      maxGSSize = 500,
      pvalueCutoff = 1,
      pAdjustMethod = "BH",
      verbose = FALSE,
      eps = 0,
      seed = TRUE
    ),
    error = function(e) {
      message("GSEA failed for ", ct, ": ", e$message)
      NULL
    }
  )

  if (is.null(gsea_ct)) next

  res_all <- as.data.frame(gsea_ct)

  if (nrow(res_all) == 0) {
    message("No GSEA result: ", ct)
    next
  }

  res_all$celltype <- ct
  res_all$direction <- ifelse(
    res_all$NES > 0,
    "LPS",
    "PBS"
  )

  gsea_list_all[[ct]] <- res_all

  gsea_ct2 <- tryCatch(
    simplify(
      gsea_ct,
      cutoff = 0.7,
      by = "p.adjust",
      select_fun = min
    ),
    error = function(e) {
      message("simplify skipped for ", ct, ": ", e$message)
      gsea_ct
    }
  )

  res_simplified <- as.data.frame(gsea_ct2)

  if (nrow(res_simplified) > 0) {

    res_simplified$celltype <- ct

    res_simplified$direction <- ifelse(
      res_simplified$NES > 0,
      "LPS",
      "PBS"
    )

    gsea_list_simplified[[ct]] <- res_simplified
  }
}
