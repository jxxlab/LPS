library(Seurat)

celltypes <- c(
  "1 M_Apoe",
  "2 M_Adgre4",
  "3 M_Cxcl2",
  "4 M_Cd209a",
  "5 pDC",
  "6 DC1",
  "7 mature DC"
)

selected_terms <- c(
  "antigen processing and presentation of exogenous peptide antigen via MHC class II",
  "positive regulation of macrophage chemotaxis",
  "canonical glycolysis",
  "cellular response to interleukin-1",
  "granulocyte chemotaxis",
  "killing by host of symbiont cells"
)

stopifnot(all(celltypes %in% pbmc4$celltype))

gsea_list <- list()

for (ct in celltypes) {

  message("Running: ", ct)

  cells_ct <- rownames(pbmc4@meta.data)[
    pbmc4$celltype == ct
  ]

  obj <- subset(pbmc4, cells = cells_ct)
  Idents(obj) <- "groups"

  n_group <- table(Idents(obj))

  if (!all(c("PBS", "LPS") %in% names(n_group)) ||
      any(n_group[c("PBS", "LPS")] < 3)) {
    warning("Skipping ", ct, ": insufficient cells")
    next
  }

  deg <- FindMarkers(
    obj,
    ident.1 = "LPS",
    ident.2 = "PBS",
    only.pos = FALSE,
    min.pct = 0,
    logfc.threshold = 0,
    return.thresh = 1
  )

  fc_col <- if ("avg_log2FC" %in% colnames(deg)) {
    "avg_log2FC"
  } else {
    "avg_logFC"
  }

  geneList <- deg[[fc_col]]
  names(geneList) <- rownames(deg)

  valid <- is.finite(geneList) &
    !is.na(names(geneList)) &
    names(geneList) != ""

  geneList <- geneList[valid]
  geneList <- sort(geneList, decreasing = TRUE)

  if (anyDuplicated(names(geneList))) {
    geneList <- geneList[!duplicated(names(geneList))]
  }

  # GO Biological Process GSEA
  gsea <- gseGO(
    geneList = geneList,
    OrgDb = org.Mm.eg.db,
    keyType = "SYMBOL",
    ont = "BP",
    minGSSize = 10,
    maxGSSize = 500,
    pvalueCutoff = 1,
    pAdjustMethod = "BH",
    verbose = FALSE,
    eps = 0
  )

  result <- as.data.frame(gsea)

  if (nrow(result) == 0) next

  result$celltype <- ct
  gsea_list[[ct]] <- result
}

if (length(gsea_list) == 0) {
  stop("No GSEA results generated.")
}

gsea_all <- bind_rows(gsea_list)

heat_df <- expand_grid(
  celltype = celltypes,
  Description = selected_terms
) %>%
  left_join(
    gsea_all %>%
       dplyr::select(celltype, Description, NES, p.adjust),
    by = c("celltype", "Description")
  ) %>%
  mutate(
    sig = case_when(
      !is.na(p.adjust) & p.adjust < 0.001 ~ "***",
      !is.na(p.adjust) & p.adjust < 0.01  ~ "**",
      !is.na(p.adjust) & p.adjust < 0.05  ~ "*",
      TRUE ~ ""
    ),
    celltype = str_remove(celltype, "^[0-9]+\\s+"),
    Description = factor(
      Description,
      levels = rev(selected_terms)
    )
  )

celltype_order <- str_remove(
  celltypes, "^[0-9]+\\s+"
)

heat_df$celltype <- factor(
  heat_df$celltype,
  levels = celltype_order
)

print(
  heat_df %>%
     dplyr::select(celltype, Description, NES, p.adjust, sig)
)

max_NES <- max(abs(heat_df$NES), na.rm = TRUE)

if (!is.finite(max_NES)) {
  stop("Selected GO terms have no matching NES values.")
}
