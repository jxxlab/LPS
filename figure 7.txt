# ----------------------------
pbmc1 <- readRDS("LPS-sub13-7474.rds")

Idents(pbmc1)="C4"
options(repr.plot.width=7, repr.plot.height=6)
DimPlot(object = pbmc1,cols = colorRampPalette(brewer.pal(12, "Paired"))(4),label = T,label.size = 8)+ theme(legend.key.height=unit(0.28,"inch"),axis.text=element_blank(),axis.ticks=element_blank(),legend.text=element_text(size=20,face="plain"),text=element_text(size=20,face="plain"))

Idents(pbmc1)="groups"

pbmc1.markers <- FindAllMarkers(object = pbmc1, only.pos = TRUE, min.pct = 0.25, logfc.threshold = 0.25)
dim(pbmc1.markers)

pbmc1x <- ScaleData(pbmc1,  features = rownames(pbmc1))

pbmc1.markers %>% group_by(cluster) %>% top_n(n = 20,  wt = avg_log2FC) -> top20

options(repr.plot.width=10, repr.plot.height=15) 
DoHeatmap(pbmc1x, features = as.character(top20$gene)) + NoLegend() +theme(text=element_text( size=24, face="bold.italic"))
